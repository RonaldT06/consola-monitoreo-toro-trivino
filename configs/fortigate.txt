# =====================================================================
#  FortiGate (FortiOS 7.x) — SNMP para NetMon
#  Servidor NetMon: 192.168.50.1 (tu PC en VMnet8) | Gestión: port1 | cambia "192.168.50" por tu red
# =====================================================================

# 1) Permitir SNMP y ping en la interfaz de gestión
config system interface
    edit "port1"
        set mode static
        set ip 192.168.50.11 255.255.255.0
        set allowaccess ping https ssh snmp
    next
end

# 1b) Ruta por defecto y DNS (para activar la licencia de prueba con FortiCare)
config router static
    edit 1
        set gateway 192.168.50.2
        set device "port1"
    next
end
config system dns
    set primary 192.168.50.2
end

# 2) Información del sistema + umbrales para traps
config system snmp sysinfo
    set status enable
    set description "FW-FORTIGATE"
    set contact-info "NOC"
    set location "Laboratorio"
    set trap-high-cpu-threshold 80
    set trap-low-memory-threshold 80
end

# 3) SNMP v2c (laboratorio). Solo el servidor NetMon puede consultar.
#    OJO: FortiGate envía los traps con esta misma community -> debe estar
#    en traps.communities de config.yaml
config system snmp community
    edit 1
        set name "NetMonRO"
        config hosts
            edit 1
                set ip 192.168.50.1 255.255.255.255
                set host-type any
            next
        end
        set query-v2c-status enable
        set trap-v2c-status enable
        set trap-v2c-rport 162
        set events cpu-high mem-low log-full intf-ip vpn-tun-up vpn-tun-down ha-switch
        # si algún evento no existe en tu versión:  set events ?
    next
end
